bash <<'VDOT_TIMING_RETRY'
#!/usr/bin/env bash
# Reuse the accepted artifacts; rerun only the unfinished timing application.
set -euo pipefail
work=/root/private_data/mj/xiangshan
cd "$work"
package="$work/evidence/vdot-timing-package-AFD1zJ/package"
emu="$work/vdot-mtval-fix-20261024/XiangShan/build/emu"
ref="$work/vdot-dev-20261024/NEMU/build/riscv64-nemu-interpreter-so"
img="$work/vdot-dev-20261024/nexus-am/apps/vdot-timing-pilot-v1/build/vdot-timing-pilot-v1-riscv64-xs.bin"
flash="$work/evidence/vdot-build-80mKCD/boot-flash.bin"
cfg="$work/vdot-dev-20261024/NEMU/.config"
run=$(mktemp -d "$work/evidence/vdot-timing-retry-XXXXXX")
trap 'rc=$?; printf "TIMING_RETRY_SCRIPT_EXIT=%s\nLOG_DIR=%s\n" "$rc" "$run" | tee "$run/script-result.txt"; exit "$rc"' EXIT
printf 'LOG_DIR=%s\n' "$run"

# 1. Check that no heavy job is active and all five runtime files are unchanged.
ps -eo pid,comm > "$run/processes-before.txt"
if awk '$2 ~ /^(emu|make|java|mill|verilator_bin)$/ {print; found=1} END {exit !found}' "$run/processes-before.txt"; then
  echo 'STOP: an existing build or simulation is running.'
  exit 3
fi
printf '%s  %s\n' \
  f9653532fd2a4b9f0d0cc08c131f7c258d4b62c46e672c6886d6643aad926850 "$emu" \
  11b9f6807c36ed860e0c64cc4e58d27216b19335ecde1ce5797ba2d1cd057b74 "$ref" \
  8545abf1afb71ec405a8c81bada3fe5054a332964436eb7976a824edaf91bda6 "$img" \
  fea548655a8b6eecacace83e2f4380a18b8aa83bce964c8aae4cffbe2f5a9724 "$flash" \
  64282bca051b5dbc5fca742f4fbca2ade937a70f3c4d2ec93c1f4836aca5d9be "$cfg" > "$run/expected.sha256"
sha256sum -c "$run/expected.sha256" | tee "$run/identity-before.txt"
# Reuse only the log/sample checking functions from the original package.
printf '%s  %s\n' \
  c5ebb06aa2fa2c338fbd2528cef862ed1eb99ea534605ccd30f06c231eab55ae "$package/run.py" \
  2f560c69880f1c243099cbebe5e1cddede0254cbc139876fcbad24848e9e46f3 "$package/failure_parser.py" \
  c5cf6bdcadec5c4203d70f5418e0959c2e1c8eda087543392d4da52c5c474195 "$package/matrix.json" > "$run/helpers.sha256"
sha256sum -c "$run/helpers.sha256" | tee "$run/helper-check.txt"

# 2. Run the same image from the beginning, with a 2400-second wall limit.
mkdir "$run/build"
export AM_HOME="$work/vdot-dev-20261024/nexus-am"
export NEMU_HOME="$work/vdot-dev-20261024/NEMU"
export NOOP_HOME="$run"
argv=("$emu" -i "$img" "--diff=$ref" -F "$flash" --seed=1 --no-snapshot --max-cycles=500000)
python3 - "$run" "${argv[@]}" <<'VDOT_RETRY_METADATA'
import json, os, sys
from pathlib import Path
out = Path(sys.argv[1])
(out/'architecture.argv.json').write_text(json.dumps(sys.argv[2:], indent=2)+'\n')
(out/'environment.json').write_text(json.dumps({k:os.environ[k] for k in ('AM_HOME','NEMU_HOME','NOOP_HOME')}, indent=2)+'\n')
VDOT_RETRY_METADATA
printf 'RUNNING_TIMING_RETRY wall_limit=2400s guest_limit=500000 seed=1 rebuild=false\n'
date -Iseconds > "$run/start.txt"
set +e
(ulimit -f 131072; timeout --signal=TERM --kill-after=5s 2400s "${argv[@]}") > "$run/architecture.log" 2>&1
emu_rc=$?
set -e
date -Iseconds > "$run/end.txt"
printf 'EMU_OR_TIMEOUT_EXIT=%s\n' "$emu_rc" | tee "$run/run-result.txt"
sha256sum -c "$run/expected.sha256" | tee "$run/identity-after.txt"

# 3. Require normal completion, all 42 samples, and the existing application checks.
python3 - "$run" "$package" "$ref" "$emu_rc" <<'VDOT_RETRY_ANALYSIS'
import importlib.util, json, sys
from pathlib import Path
sys.dont_write_bytecode = True
out, package = map(Path, sys.argv[1:3])
ref, rc = sys.argv[3], int(sys.argv[4])
summary = {'status':'failed', 'cwd':str(Path.cwd()), 'emu_or_timeout_exit':rc, 'wall_limit_seconds':2400,
           'guest_cycle_limit':500000, 'seed':1, 'rebuild':False,
           'final_speedup_accepted':False, 'scope':'raw serialized timing interval; includes measurement overhead'}
def save(name, data):
    (out/name).write_text(json.dumps(data, indent=2)+'\n')
try:
    sys.path.insert(0, str(package))
    spec = importlib.util.spec_from_file_location('pilot_checker', package/'run.py')
    checker = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(checker)
    with (out/'architecture.log').open(errors='replace') as f:
        analysis = checker.inspect(f, ref, {})
    save('architecture-check.json', analysis)
    checker.require(rc == 0, 'emu/timeout exit is '+str(rc)+'; inspect architecture.log')
    checker.require(analysis['reference'] and analysis['difftest'] and analysis['good_trap'] and
                    analysis['arch_marker'] and not analysis['errors'] and
                    not analysis['usage'] and not analysis['limit'], 'Runtime acceptance markers failed')
    matrix = json.loads((package/'matrix.json').read_text())
    with (out/'architecture.log').open(errors='replace') as f:
        rows, stats = checker.inspect_samples(f, matrix)
    checker.require([(r['round'],r['position']) for r in rows] ==
                    [(r,p) for r in range(7) for p in range(6)], 'Sample order differs')
    save('samples.json', rows)
    save('raw-statistics.json', stats)
    (out/'samples.csv').write_text('warm,round,position,variant,cycles,instret,consumer\n'+
        ''.join(','.join(str(r[k]) for k in ('warm','round','position','variant','cycles','instret','consumer'))+'\n' for r in rows))
    summary.update(status='timing-pilot-raw-results-review-required', calls=42, warmups=6, samples=36)
    with (out/'architecture.log').open(errors='replace') as f:
        for line in f:
            if any(s in line for s in ('The reference model is ', 'Difftest enabled.', 'PASS VDOT_TIMING_PILOT', 'HIT GOOD TRAP')):
                print(line.strip())
    for stat in stats:
        print('PILOT_RAW='+json.dumps(stat))
    print('VDOT_TIMING_PILOT_READY samples=36 warmups=6')
except Exception as error:
    summary['error'] = str(error)
    print('TIMING_RETRY_ERROR='+str(error))
finally:
    save('result.json', summary)
    print('LOG_DIR='+str(out))
sys.exit(0 if summary['status']=='timing-pilot-raw-results-review-required' else 2)
VDOT_RETRY_ANALYSIS

VDOT_TIMING_RETRY
